import android.content.ComponentName;
import android.content.Intent;
import android.content.pm.ActivityInfo;
import android.content.pm.ResolveInfo;
import android.net.Uri;
import android.os.Bundle;
import android.webkit.JavascriptInterface;
import android.webkit.WebResourceRequest;
import android.webkit.WebView;
import com.getcapacitor.BridgeActivity;
import com.getcapacitor.BridgeWebViewClient;
import java.util.List;

public class MainActivity extends BridgeActivity {

  class Opener {
    @JavascriptInterface
    public boolean open(final String pkg, final String uri) {
      try {
        getPackageManager().getPackageInfo(pkg, 0);
      } catch (Exception e) {
        return false;
      }
      runOnUiThread(new Runnable() {
        @Override
        public void run() {
          try {
            Intent i;
            if (uri != null && uri.length() > 0) {
              i = new Intent(Intent.ACTION_VIEW, Uri.parse(uri));
              i.setPackage(pkg);
            } else {
              i = getPackageManager().getLaunchIntentForPackage(pkg);
            }
            if (i != null) {
              i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
              startActivity(i);
            }
          } catch (Exception ignored) {}
        }
      });
      return true;
    }
  }

  @Override
  public void onCreate(Bundle savedInstanceState) {
    super.onCreate(savedInstanceState);
    getBridge().getWebView().addJavascriptInterface(new Opener(), "AndroidApp");
  }

  @Override
  public void onStart() {
    super.onStart();
    getBridge().getWebView().setWebViewClient(new BridgeWebViewClient(getBridge()) {
      @Override
      public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest req) {
        String url = req.getUrl().toString();
        if (url.startsWith("intent:")) {
          try {
            Intent i = Intent.parseUri(url, Intent.URI_INTENT_SCHEME);
            i.setSelector(null);
            if (i.getPackage() != null) {
              List<ResolveInfo> l = getPackageManager().queryIntentActivities(i, 0);
              if (!l.isEmpty()) {
                ActivityInfo ai = l.get(0).activityInfo;
                i.setComponent(new ComponentName(ai.packageName, ai.name));
              }
            }
            startActivity(i);
          } catch (Exception e) {
            try {
              String fb = Intent.parseUri(url, Intent.URI_INTENT_SCHEME).getStringExtra("browser_fallback_url");
              if (fb != null) startActivity(new Intent(Intent.ACTION_VIEW, Uri.parse(fb)));
            } catch (Exception ignored) {}
          }
          return true;
        }
        return super.shouldOverrideUrlLoading(view, req);
      }
    });
  }
}
